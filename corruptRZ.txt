local selectedTarget = "All Players"
local function getPlayer(target)
	local found = {}
	if not target then return found end
	if target == "All Players" then
		return Players:GetPlayers()
	elseif target == "LocalPlayer (Me)" then
		return {lp}
	else
		for _, p in pairs(Players:GetPlayers()) do
			if p.Name == target or p.DisplayName == target then
				table.insert(found, p)
			end
		end
	end
	return found
end

local function fireRemoteSafe(remote, args, retries)
	if not remote then return false end
	retries = retries or 3
	for i = 1, retries do
		local success, err = pcall(function()
			if remote:IsA("RemoteEvent") then
				remote:FireServer(unpack(args or {}))
			elseif remote:IsA("RemoteFunction") then
				remote:InvokeServer(unpack(args or {}))
			end
		end)
		if success then return true end
		task.wait(0.1)
	end
	return false
end

local function getZombieHealth(zombie)

local function isZombieVisible(zombie)
	if not zombie or not zombie:FindFirstChild("HumanoidRootPart") or not Workspace.CurrentCamera then return false end
	local camPos = Workspace.CurrentCamera.CFrame.Position
	local targetPos = zombie.HumanoidRootPart.Position
	local raycastParams = RaycastParams.new()
	raycastParams.FilterType = Enum.RaycastFilterType.Exclude
	raycastParams.FilterDescendantsInstances = {getChar(), zombie}
	
	local result = Workspace:Raycast(camPos, targetPos - camPos, raycastParams)
	return result == nil
end

local function getClosestZombie(range, partName, visCheck)
	local closest, shortestDist = nil, range or math.huge
	local hrp = getChar() and getChar():FindFirstChild("HumanoidRootPart")
	if not hrp then return nil end

	for _, z in pairs(zombieGameData["zombies"]()) do
		local targetPart = z:FindFirstChild(partName or "Head") or z:FindFirstChild("HumanoidRootPart")
		if targetPart and z:FindFirstChildOfClass("Humanoid") and z:FindFirstChildOfClass("Humanoid").Health > 0 then
			if not visCheck or isZombieVisible(z) then
				local dist = (targetPart.Position - hrp.Position).Magnitude
				if dist < shortestDist then
					shortestDist = dist
					closest = z
				end
			end
		end
	end
	return closest
end

-- Dynamic Shooting Remote Engine
local dynamicGunRemote = nil
local dynamicShootString = nil

local function shootZombieToAttach(zombie, partName)
	if not zombie then return false end
	local targetPart = zombie:FindFirstChild(partName or "Head") or zombie:FindFirstChild("HumanoidRootPart")
	if not targetPart then return false end

	-- Method 1: Dynamically Captured Lighting Remote / Hit_Object
	if dynamicGunRemote or zombieGameData["Paths"]["ShootGun"] then
		local rem = dynamicGunRemote or zombieGameData["Paths"]["ShootGun"]
		if rem and rem:IsA("RemoteEvent") then
			local str = dynamicShootString or "Hit"
			rem:FireServer(targetPart, targetPart.Position, str)
			rem:FireServer({targetPart, targetPart.Position, zombie:FindFirstChildOfClass("Humanoid")})
		end
	end

	-- Method 2: AttemptWeaponHit Core Remote
	if R.WeaponHit then
		R.WeaponHit:FireServer(zombie, targetPart, targetPart.Position)
	end

	-- Method 3: Standard Engine Fallback Damage Path
	local hum = zombie:FindFirstChildOfClass("Humanoid")
	if hum and zombieGameData["Paths"]["Damage"] then
		local strKey = zombieGameData["Strings"]["Damage"] or "Damage5\240\159\144\153"
		local args = {
			{
				hum,
				math.huge,
				strKey,
				false, false, false,
				[8] = targetPart.Position,
				[9] = { HitS = 1, ConfirmedHit = 1 }
			}
		}
		zombieGameData["Paths"]["Damage"]:FireServer(unpack(args))
	end
	return true
end

	oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
		local method = getnamecallmethod():lower()
		local args = {...}
		
		-- Dynamic Capture for Gun Shooting Remotes
		if not checkcaller() and method:find("fire") then
			if self.Name:lower():find("hit") or self.Name:lower():find("shoot") or self == zombieGameData["Paths"]["ShootGun"] then
				dynamicGunRemote = self
				if args[3] and type(args[3]) == "string" then
					dynamicShootString = args[3]
				end
			end
		end

		-- Capture Remote Strings
		if not checkcaller() and method:find("fire") and self == damagerem then
			if args[1] and type(args[1]) == "table" and args[1][1] and args[1][3] then
				local strArg = args[1][3]
				if type(strArg) == "string" and not tonumber(strArg) then
					if strArg:lower():find("rewards") then
						rawset(zombieGameData["Strings"], "Rewards", strArg)
					elseif strArg:lower():find("damage") then
						rawset(zombieGameData["Strings"], "Damage", strArg)
					end
				end
			end
		end

		-- Zombie Damage Block Hook
		if noDamageToLocal and method:find("fire") and self == damagerem then
			if args[1] and type(args[1]) == "table" then
				local targetHum = args[1][1]
				if getChar() and (targetHum == getChar() or (targetHum:IsA("Humanoid") and targetHum.Parent == getChar())) then
					local dmgVal = args[1][2]
					if type(dmgVal) == "number" and dmgVal > 0 then
						return -- Block damaging calls
					end
				end
			end
		end
		
		return oldNamecall(self, ...)
	end)

	-- Shoot Through Walls / Raycast Override
	local oldRaycast
	oldRaycast = hookfunction(Workspace.Raycast, function(self, origin, direction, params, ...)
		if wallbangEnabled and not checkcaller() then
			if not params then
				params = RaycastParams.new()
			end
			params.FilterType = Enum.RaycastFilterType.Exclude
			
			local ignoreTable = params.FilterDescendantsInstances or {}
			table.insert(ignoreTable, Workspace:FindFirstChild("Map"))
			table.insert(ignoreTable, Workspace:FindFirstChild("MapGeometry"))
			table.insert(ignoreTable, Workspace:FindFirstChild("Structure"))
			table.insert(ignoreTable, Workspace:FindFirstChild("Barriers"))
			
			params.FilterDescendantsInstances = ignoreTable
		end
		return oldRaycast(self, origin, direction, params, ...)
	end)

	local timeout = 0
	while timeout < 60 do
		if zombieGameData["Strings"]["Rewards"] ~= nil and zombieGameData["Strings"]["Damage"] ~= nil then
			break
		end
		task.wait(0.1)
		timeout = timeout + 1
	end

	if zombieGameData["Strings"]["Rewards"] and zombieGameData["Strings"]["Damage"] then
		createnotification("Auto-Updater", "Keys captured successfully!", 4)
	else
		createnotification("Auto-Updater", "Applied fallback remote keys.", 4)
		zombieGameData["Strings"]["Rewards"] = zombieGameData["Strings"]["Rewards"] or "Rewards8\240\159\144\153"
		zombieGameData["Strings"]["Damage"] = zombieGameData["Strings"]["Damage"] or "Damage5\240\159\144\153"
	end
else
	createnotification("Auto-Updater", "Hookmetamethod unavailable! Standard fallbacks enabled.", 5)
	zombieGameData["Strings"]["Rewards"] = "Rewards8\240\159\144\153"
	zombieGameData["Strings"]["Damage"] = "Damage5\240\159\144\153"
end

local function resPlr(plr)
	if plr == lp then 
		if R.ForceRevive then
			R.ForceRevive:FireServer()

		end
		if getChar():FindFirstChild("Downed") then
			getChar():FindFirstChild("Downed"):Destroy()
		end

end

-- Machine & Switch Finder
local function getMachine(machine)
	if not interactions then return nil end
	if machine == "mystery" then
		local box = interactions:FindFirstChild("Mystery")
		return box and box:FindFirstChild("Shop") or nil
	end
	for _, v in pairs(interactions:GetChildren()) do
		if v.Name:lower():find(machine:lower()) then
			local model = v:FindFirstChild("Shop Place") or v:FindFirstChild("Shop")
			if not model then continue end
			if model:IsA("Model") or model:IsA("BasePart") then
				model = model:FindFirstChild("Shop") or model
				if model:IsA("Model") and model:GetPivot().Y > 9000 then continue end
				if model:IsA("BasePart") and model.Position.Y > 9000 then continue end
				return model
			end
		end
	end
	return nil
end

local function triggerPowerSwitches()
	local triggered = 0
	if interactions then
		for _, v in pairs(interactions:GetChildren()) do
			local n = v.Name:lower()
			if n:find("power") or n:find("lever") or n:find("switch") or n:find("generator") then
				local interactRemote = v:FindFirstChild("Interact", true) or v:FindFirstChild("Shop", true)
				if interactRemote and interactRemote:IsA("RemoteEvent") then
					interactRemote:FireServer()
					triggered = triggered + 1
				end
			end
		end
	end
	if R.PowerSwitch then
		R.PowerSwitch:FireServer(true)
		triggered = triggered + 1
	elseif ReplicatedStorage:FindFirstChild("Events") then
		for _, rem in pairs(ReplicatedStorage.Events:GetChildren()) do
			local n = rem.Name:lower()
			if n:find("power") or n:find("switch") or n:find("turnon") then
				rem:FireServer(true)
				triggered = triggered + 1
			end
		end
	end
	return triggered
end

-- Waypoint Setup Engine
local WaypointsTab
local function wp(name, obj)
	if not name or not obj or not WaypointsTab then return end
	local pos = obj:IsA("Model") and obj:GetPivot().Position or obj.Position
	local posStr = pos.X..", "..pos.Y..", "..pos.Z
	local data = { ["pos"] = posStr }

	zombieGameData["Dropdowns"][name] = WaypointsTab:CreateDropdown({
		Name = "WP: " .. name,
		Options = {"teleport to", "cancel", "loop tp", "unloop tp", "anticheat goto"},
		CurrentOption = {"actions"},
		MultipleOptions = false,
		Flag = "WP_" .. name,
		Callback = function(Option)
			local value = parseOption(Option)
			if value == "cancel" or value == "" or value == "actions" then return end
			local postable = data['pos']:split(", ")
			local cf = CFrame.new(tonumber(postable[1]), tonumber(postable[2]), tonumber(postable[3]))

			if value == "teleport to" then
				getChar():PivotTo(cf)
			elseif value == "loop tp" then
				stoploop("looptping")
				loopConnections["looptping"] = RunService.Heartbeat:Connect(function()
					getChar():PivotTo(cf)
				end)
			elseif value == "unloop tp" then
				stoploop("looptping")
			elseif value == "anticheat goto" then
				local part = Instance.new("Part", Workspace)
				part.Anchored = true
				part.Transparency = 1
				part.CanCollide = false
				part.CFrame = cf
				tpBypass(part)
				part:Destroy()
			end
		end,
	})
end

		end

		-- Try the "Kills" source pattern with double XP flag
		R.XPGain:FireServer(xpAmount, "Kills", doubleXPToggle)

		createnotification(
			"XP Gain",
