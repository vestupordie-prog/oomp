local function fireRemoteSafe(remote, args, retries)
	if not remote then return false end
	retries = retries or 3
	for i = 1, retries do
		local success, err = pcall(function()
			if remote:IsA("RemoteEvent") then
				remote:FireServer(unpack(args or {}))
			elseif remote:IsA("RemoteFunction") then
				remote:InvokeServer(unpack(args or {}))
			end
		end)
		if success then return true end
		task.wait(0.1)
	end
	return false
end

local function isZombieVisible(zombie)
	if not zombie or not zombie:FindFirstChild("HumanoidRootPart") or not Workspace.CurrentCamera then return false end

	end
	return closest

	if not name or not obj or not WaypointsTab then return end
	local pos = obj:IsA("Model") and obj:GetPivot().Position or obj.Position
	local posStr = pos.X..", "..pos.Y..", "..pos.Z

				part.Anchored = true

				part.CanCollide = false
				part.CFrame = cf

-- Helper builder for 60+ remotes tab expansion
local function addRemoteControlBlock(tab, remoteObj, remoteName)
	tab:CreateSection(remoteName)
	tab:CreateButton({
		Name = "Fire " .. remoteName,
		Callback = function()
			fireRemoteSafe(remoteObj, {})
			notifyLog("Remote Engine", "Fired " .. remoteName)
		end,
	})
	
	local loopState = false
	tab:CreateToggle({
		Name = "Fire Loop: " .. remoteName,
		CurrentValue = false,
		Callback = function(val)
			loopState = val
			if val then
				loopConnections["remLoop_" .. remoteName] = RunService.Heartbeat:Connect(function()
					fireRemoteSafe(remoteObj, {})
				end)
			else
				stoploop("remLoop_" .. remoteName)
			end
		end,
	})
	
	local argsInputText = "{}"
	tab:CreateInput({
		Name = "Fire With Args (JSON) - " .. remoteName,
		PlaceholderText = "[\"Arg1\", 100]",
		RemoveTextOnFocusLost = false,
		Callback = function(txt) argsInputText = txt end,
	})
	
	tab:CreateButton({
		Name = "Fire " .. remoteName .. " With Args",
		Callback = function()
			local parsedArgs = {}
			pcall(function() parsedArgs = HttpService:JSONDecode(argsInputText) end)
			fireRemoteSafe(remoteObj, parsedArgs)
			notifyLog("Remote Engine", "Fired " .. remoteName .. " with custom arguments!")
		end,
	})
end
